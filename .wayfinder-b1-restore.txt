--- comment: ## 外置控制条实施规格（#228 七问 → #227 实施输入）

规格按待固化清单 1–10 逐项落定。除注明外，坐标一律 DIP（Electron bounds 原生口径，负坐标天然成立）。

### 1. 第二控制条窗口（bar）拓扑

- 新增 `barWin = new BrowserWindow({ frame:false, transparent:true, alwaysOnTop:true, skipTaskbar:true, resizable:false, hasShadow:false, focusable:false, show:false })`；`ready-to-show` 后定一次几何再 show，此后**常驻，不再 show/hide**。webPreferences 与主浮窗同口径（preload `dist/preload/preload.cjs`、contextIsolation:true、nodeIntegration:false；preload 零改动复用）。
- **focusable=false 定案**：点击 bar 不抢焦点（不激活窗口、仍收鼠标），播放控制不打断看片；主浮窗 focusable 维持现状（零行为差异）。
- 装载 `dist/renderer/bar.html + bar.js`（经典脚本、import-free，同 renderer.ts 约束；tsconfig.build `src/**/*` 自动发出；build.mjs 增一行 bar.html 拷贝）。
- 创建时机：随 `startProductStack()`；scene 骨架模式（SKELETON_SCENE）不创建；`YOUTUBESUB_E2E_OFFSCREEN=1` 时与主浮窗同门控 `show:false`。
- **显隐 = opacity + ignore 双门控**：不可见态由 ① renderer `opacity:0` 过渡 + ② main 侧 `barWin.setIgnoreMouseEvents(true,{forward:true})`。理由：淡出宽限期需要窗在场，show/hide 会掐掉过渡；ignore+forward 让不可见态对桌面零干扰（forward 合成 enter/leave 正是桥接依赖的路径）。
- z 序：与主浮窗同为 alwaysOnTop；5s topmost 脉冲改双窗（先浮窗后 bar，各 `setAlwaysOnTop(true)+moveTop()`）。bar 与浮窗不重叠（GAP=8，正贴/翻转两种位形都不叠），相对 z 序无感。

### 2. 跟随几何（移动 / resize / 翻转 / 多显示器 / 负坐标）

- **事件源（单一）**：监听主浮窗 `move` + `resize`（拖拽 / 8 边 resize 循环里的 setPosition/setBounds 天然发出；persistGeometry 落盘不变），加 `screen.on('display-metrics-changed')` 兜底（拔插显示器 / 改分辨率 / 托盘重显后重定位）。**不节流**：实测 setPosition 492µs，与浮窗拖拽同频各一次在 #212 CPU 预算内；复测超闸的升级路径 = rAF 合帧，先不做。

407	      if (i === 0 && rows.length === 2) events.push({ kind: "divider", y: 20 * i + 2 });
408	    });
409	    return { events, status_text: this.status_text, bg, anim: this.anim_vars(),
410	      state: this.overlay_state, title: this.title_text,
411	      playing: this.playing, position_ms: this.position_ms,
412	      duration_ms: this.duration_ms, muted: this.muted };
413	  }
414	
415	  // CSS custom properties for the animation tuning area (#222): read per
416	  // paint from the anim.* section, garbage values fall back to the defaults
417	  // the same way the tuning table does (old configs without the keys are
418	  // normal, like the #174 appearance keys).
419	  anim_vars(): AnimVars {
420	    const a = this.settings["anim"];
421	    const node = (a !== null && typeof a === "object" && !Array.isArray(a)
422	      ? a : {}) as { [k: string]: unknown };
423	    const numOr = (key: "fade_delay" | "dur" | "shrink"): number => {
424	      const v = node[key];
425	      return typeof v === "number" && Number.isFinite(v) ? v : FALLBACK_ANIM_NUMS[key];
426	    };
427	    const cssNum = (v: number): string => String(Math.round(v * 1000) / 1000);
428	    const ease = node["ease"];
429	    return {
430	      fade_delay: cssNum(numOr("fade_delay")) + "s",
431	      dur: cssNum(numOr("dur")) + "s",
432	      shrink: cssNum(numOr("shrink")),
433	      ease: typeof ease === "string" && (ANIM_EASES as readonly string[]).includes(ease)
434	        ? ease : FALLBACK_ANIM_EASE,
435	    };
436	  }
437	
438	  // ---- three-state machine (#211 decision block, #217) ----
439	  //
440	  // Derived state, recomputed after every input change:
441	  //   click_through ? passthrough : frozen ? keep : hover ? controls : subtitles.
442	  // freeze (drag/8-edge resize in progress, context menu open) gates ONLY the
443	  // automatic enter/leave transitions; explicit passthrough flips are user
444	  // actions and always land. Unlock (click-through off) clears hover so the
445	  // overlay falls back to subtitles, never straight into controls (#211:
446	  // 解锁图标点击 → subtitles，不落 controls). Timing is NOT this machine's
447	  // business: a state flip is instant here, and the fade-delay/shrink numbers
448	  // ride anim_vars into CSS transitions (#214 matrix).
449	  private overlay_recompute(): void {
450	    this.overlay_state = this.overlay_click_through
451	      ? "passthrough"
452	      : this.overlay_frozen
453	        ? this.overlay_state
454	        : (this.overlay_hover ? "controls" : "subtitles");
455	  }
456	
457	  set_overlay_hover(hovering: boolean): void {
458	    this.overlay_hover = hovering;
459	    this.overlay_recompute();
460	  }
461	
462	  set_overlay_frozen(frozen: boolean): void {
463	    this.overlay_frozen = frozen;
464	    this.overlay_recompute();
465	  }
466	
467	  set_overlay_click_through(on: boolean): void {
468	    this.overlay_click_through = on;
469	    if (!on) this.overlay_hover = false; // unlock lands on subtitles
470	    this.overlay_recompute();
471	  }
472	
473	  // ---- settings reread (#152 / spec #161) ----
474	  reread_settings(): void {
475	    // Re-read the two snapshotted display choices: the tuning page calls this
476	    // after 确定 so the overlay changes face without a restart. History,
477	    // texts, geometry and click-through are left alone. Never saves.
478	    const disp = dispSection(this.settings);
479	    this.mode = asText(disp["mode"]) || "bilingual";
480	    this.order = asText(disp["order"]) || "trans_first";
481	  }
482	
483	  // ---- menu-driven mutations ----
484	  cycle_mode(): void {
485	    // Garbage mode coerced to the first slot instead of Python's ValueError.
486	    const i = Math.max(0, OVERLAY_MODES.indexOf(this.mode as (typeof OVERLAY_MODES)[number]));
487	    this.mode = OVERLAY_MODES[(i + 1) % OVERLAY_MODES.length]!;
488	    dispSection(this.settings)["mode"] = this.mode; // in-memory only, like Python (no save)
489	  }
490	
491	  swap_order(): void {
492	    // Exact Python ternary: anything but trans_first lands back on trans_first.
493	    this.order = this.order === "trans_first" ? "orig_first" : "trans_first";
494	    dispSection(this.settings)["order"] = this.order; // in-memory only, like Python (no save)
495	  }

497	  private nudgeTuned(key: string, delta: number): number {
498	    // 菜单里的 ± 快捷方式：范围取权威表（TUNING_FIELDS），**改完即落盘** ——
499	    // 这两项是从外面改浮窗外观的唯一途径，不能等下次挪窗口才顺手存进去
500	    // （#167 决议）。
501	    const field = this.fieldSpec(key);
502	    const low = Math.trunc(field.min);
503	    const high = Math.trunc(field.max);
504	    const disp = dispSection(this.settings);
505	    const current = Math.trunc(num(disp[key], field.default));
506	    const value = Math.max(low, Math.min(high, current + delta));
507	    disp[key] = value;
508	    try {
509	      this.saveImpl(this.settings);
510	    } catch {
511	      // OSError: pass (Python semantics)
512	    }
513	    return value;
514	  }

525	// ---------------------------------------------------------------------------
526	// Wrap algorithm (draw_wrapped, measurer-injected)
527	// ---------------------------------------------------------------------------
528	
529	export interface Measurer {
530	  // QFontMetrics.horizontalAdvance equivalent (any consistent unit).
531	  advance(text: string): number;
532	  // QFontMetrics.height equivalent.
533	  height(): number;
534	}
535	
536	export interface Rect { x: number; y: number; w: number; h: number }
537	export interface Pt { x: number; y: number }
538	
539	export function wrap_text(text: string, area: Rect, m: Measurer, startY: number): { lines: string[]; end_y: number } {
540	  const line_h = m.height() + 3;
541	  // Wrap on spaces where possible, then split overlong words by character
542	  // width. Chinese subtitles commonly contain no spaces at all.
543	  const words = text.split(" ");
544	  const lines: string[] = [];
545	  let cur = "";
546	
547	  const split_word = (word: string): string[] => {
548	    const parts: string[] = [];
549	    let part = "";
550	    for (const ch of word) {
551	      const candidate = part + ch;
552	      if (part && m.advance(candidate) > area.w) {
553	        parts.push(part);
554	        part = ch;
555	      } else {
556	        part = candidate;
557	      }
558	    }
559	    if (part) parts.push(part);
560	    return parts;
561	  };
562	
563	  for (const w of words) {
564	    const cand = (cur + " " + w).trim();
565	    if (m.advance(cand) <= area.w) {
566	      cur = cand;
567	    } else {
568	      if (cur) {
569	        lines.push(cur);
570	        cur = "";
571	      }
572	      const parts = split_word(w);
573	      if (parts.length) {
574	        for (const p of parts.slice(0, -1)) lines.push(p);
575	        cur = parts[parts.length - 1]!;
576	      }
577	    }
578	  }
579	  if (cur) lines.push(cur);
580	
581	  // The draw loop: vertical budget with the +8 bottom tolerance, 1:1 with
582	  // _draw_wrapped. Returns only the lines that would be drawn plus end_y.
583	  const drawn: string[] = [];
584	  let y = startY;
585	  for (const ln of lines) {
586	    if (y + line_h > area.y + area.h + 8) break;
587	    drawn.push(ln);
588	    y += line_h;
589	  }
590	  return { lines: drawn, end_y: y + 4 };
591	}

593	// ---------------------------------------------------------------------------
594	// Geometry math (drag + 8-edge resize, LiveSubs-style manual calculation)
595	// ---------------------------------------------------------------------------
596	
597	export type ResizeEdge = "t" | "b" | "l" | "r" | "tl" | "tr" | "bl" | "br";
598	
599	export function edge_at(px: number, py: number, w: number, h: number): ResizeEdge | null {
600	  const m = RESIZE_MARGIN;
601	  const left = px <= m;
602	  const right = px >= w - m;
603	  const top = py <= m;
604	  const bottom = py >= h - m;
605	  if (top && left) return "tl";
606	  if (top && right) return "tr";
607	  if (bottom && left) return "bl";
608	  if (bottom && right) return "br";
609	  if (top) return "t";
610	  if (bottom) return "b";
611	  if (left) return "l";
612	  if (right) return "r";
613	  return null;
614	}
615	
616	export function apply_resize(edge: ResizeEdge, g: Pt, pressGeom: Rect, pressGlobal: Pt): Rect {
617	  // min 1x1: guard zero-size only; no user-facing limit. Baseline is the
618	  // PRESS geometry (not current) - re-applying with the same cursor position
619	  // returns to the press-time size, which is what the latch regression pins.
620	  const min_w = 1, min_h = 1;
621	  const geo: Rect = { ...pressGeom };
622	  const dx = g.x - pressGlobal.x;
623	  const dy = g.y - pressGlobal.y;
624	  if (edge.includes("l")) {
625	    const new_w = geo.w - dx;
626	    if (new_w >= min_w) geo.x = geo.x + dx;
627	  }
628	  if (edge.includes("r")) {
629	    geo.w = Math.max(min_w, geo.w + dx);
630	  }
631	  if (edge.includes("t")) {
632	    const new_h = geo.h - dy;
633	    if (new_h >= min_h) geo.y = geo.y + dy;
634	  }
635	  if (edge.includes("b")) {
636	    geo.h = Math.max(min_h, geo.h + dy);
637	  }
638	  return geo;
639	}
